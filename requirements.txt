# Upper bounds are deliberate. Without them a fresh build resolves ">=" to
# whatever shipped this morning and compiles against a version nobody has
# run the tests on. Raise these on purpose, after running the tests.
#
# No plotting library: the plan view is drawn on a plain canvas, which is
# why this starts in about a second and the download is a few MB rather
# than fifty.
customtkinter>=5.2.2,<6
# Map pictures (1.10035): the plan view drawn to a PNG. Pillow ships
# FreeType in its wheel, so it needs no Ghostscript and no system fonts.
# 10.1 is the first with a sized default font; 13 is not tested.
Pillow>=10.1,<13
# Linux (1.10039): the work keys are grabbed on the X server through
# python-xlib. Pure Python, Linux only - pip skips the line everywhere else.
# EDSMT runs without it; the keys are then off and Settings says why.
python-xlib>=0.33,<1; sys_platform == "linux"
